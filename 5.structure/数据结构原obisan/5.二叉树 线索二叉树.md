// 遍历二叉树

/*

     依次访问二叉树的左子树 跟结点 右子树

     L遍历左子树  D访问根节点  R访问右子树  则遍历方案有:

          DLR LDR LRD DRL RDL RLD 六种

  

     先根遍历(先序遍历): 先访问根节点 左子树 右子树

     中序遍历: 从左子树最左边的结点开始 先访问它的孩子 再访问它的双亲 一直到根节点 再从右子树同样的位置开始

     后序遍历: 从叶子结点开始 访问到根结点 再从另一个方向的叶子开始  

     先序遍历:

          判断是否为空子树 否则:

          (1) 访问根节点       每一次遍历都要在访问一遍当前位置的根

          (2) 先序遍历左子树   左边所有的结点都访问完了再访问右边的元素

          (3) 先序遍历右子树

  

  

     已知先序和中序序列求二叉树

          先序: A *B C D E F G* H I J

          中序: C D B* F E A I H G* J

          已知A为根 CDBFE为左子树  IHGJ为右子树

          且左子树中B为根 右子树中G为根

          整个树最左边元素为C  最右边元素为J

                         A

                    B          G

                 C     E     H   J

                  D   F     I

  
  

     已知中序和后序求二叉树

          中序: B D C E A F H G

          后序: D E C B* H G F* A

          已知A为根 BDCE为左子树   FHG为右子树

          左子树中B为根 右子树中F为根

  

                         A

                    B         F

                       C        G

                      D  E        H

*/

// 中序遍历二叉树非递归算法

/*

     在中序遍历过某个结点的在整个左子树后 如何找到该结点的根以及右子树

  

     基本思想:

          (1) 建立一个栈

          (2) 根结点进栈 遍历左子树

          (3) 根结点出栈 输出根结点 遍历右子树

  

          遍历左子树 经过结点 但不访问 一直到叶子结点 访问该结点的双亲(出栈)

          然后访问叶子结点(出栈) 访问双亲(出栈) 访问右子树

*/
```
Status lnOrderTraverse(BiTree T){

     BiTree p;

     lnitStack(S);   // 创建一个栈

     p = T;

  

     while(p || !StackEmpty(S)){

          if(p){   // 左子树

               Push(S, q);

               printf("%c", q->data);

          }

  

          else{    // 右子树

               Pop(S, q);   // q指向当前根结点

               printf("%c", q->data);

               p = q->rchild;

          }

     } // while


     return OK;
}
```

  

// 二叉树的层次遍历算法

/* 横向顺序访问 从左到右 一层一层访问 且每个结点都只访问一次

     算法思路: 使用一个队列

          (1) 将根结点进队

          (2) 队不空时循环 从队列中出列一个结点*p 访问它

               1. 若它有左孩子结点 将左孩子结点进队

               2. 若它有右孩子结点 将右孩子结点进队

*/

```
void LevelOrder(BTNode *b){

     BTNode *p;

     SqQueue *qu;

     lnitQueue(qu);             // 初始化队列

  

     enQueue(qu, b);            // 根结点指针进入队列

  

     while(!QueueEmpty(qu)){    // 队不为空 则循环

          deQueue(qu, p);       // 出队结点p

          printf("%c", p->data);      // 访问结点p

          if(p -> lchild != NULL) enQueue(qu, p->lchild);   // 有左孩子时将其进队

          if(p -> rchild != NULL) enQueue(qu, p->rchild);   // 有右孩子时将其进队

     }
}
```

/* 2.计算二叉树的深度

     如果是空树 则深度为0

     不是空树 递归计算左子树的深度记为m 递归计算右子树的深度记为n

             二叉树的深度则为m与n的较大者加1

*/
```
int Depth(BiTree T){

     if(T == NULL)  return 0;   // 如果是空树 返回0

     else{

          m = Depth(T -> lChild);   // 左子树的深度

          n = Depth(T -> rChild);   // 右子树的深度

          if(m>n) return (m+1);    

          else return (n+1);        // 取左右子树深度的最大值  +1根结点

     }

}
```

int NodeCount(BiTree T){

}

// 线索二叉树

/* 利用二叉链表中的空指针域

     如果某个结点的左孩子为空 则将空的左孩子指针域改成指向其前驱

     如果某个结点的右孩子为空 则将空的右孩子指针域改成指向其后继

     这种改变指向的指针称为线索

     加上了线索的二叉树称为线索二叉树

  

     对二叉树按某种遍历次序使其变为线索二叉树的过程叫线索化

  

     eg. 已知二叉树       A

                      B      

                   C      D

                        E    F

                         G

          对这个二叉树中序遍历  CBEGDFA

  

          依次对每个结点加线索

               A 没有右孩子 指向后继 A结点没有后继 空^

               B 无空

               C 叶子结点 没有左孩子也没有右孩子 C结点没有前驱 前驱为^ 后继指向它的前驱B

               D 无空

               E 左孩子空 指向前驱B

               F 无左孩子也无右孩子 前驱指向D 后继指向A

               G 无左孩子也无右孩子 前驱指向E 后继指向D

  

          就是让空的孩子域指向其在中序的前驱和后继

          首元素和尾元素分别没有前驱和后继 因此为空

  

  

     为区分指针域指向下方元素 还是被线索化了 对二叉链表每个结点增设两个标志域 ltag和rtag

          ltag = 0   lchild指向该结点的左孩子

          ltag = 1   lchild指向该结点的前驱

          rtag = 0   rchild指向该结点的右孩子

          rtag = 1   rchild指向该结点的后继

  

     为了避免首尾指针指向空 我们增设头结点

          ltag = 0, lchild指向根结点

          rtat = 0, rchild指向遍历序列中的最后一个结点

          遍历序列中第一个结点的lc域和最后一个结点的rc域都指向头结点

*/

  
  
  

// 定义线索二叉树
```
typedef struct BiThrNode{

     int data;

     int ltag, rtag;

     struct BiThrNode *lchild, *rchild;

} BiThrNode, *BiThrTree;
```

/* 2.孩子链表

     相当于每一行为一个整体  

     把每个结点的孩子结点排列起来 看成是一个线性表 用单链表存放

     n个结点有n个孩子链表
