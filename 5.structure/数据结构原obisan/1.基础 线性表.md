   例 两个n*n矩阵的算法可描述为
```
   for (i = 1; i <= n; i++) // n+1次 第n+1次进行判断 判断失败结束循环 但算一次
     { 
          for (j = 1; j <= n; j++) // 第n+1次进行了外层循环 但不会进行内层循环 一共n(n+1)次

          {    
               c[i][j] = 0; // 二维数组 由0开始 每次循环增1 至n.  n*n次

               for (k = 0; k < n; k++) // 三维数组  n*n*(n+1)次
               {                                                             
                    c[i][j] = c[i][j] + a[i][k] * b[k][j] // n*n*n次
               }
          }
     };

     // 计算结果可得 消耗总时间(频率)Tn = 2n³+3n²+2n+1  ->  O(n³)
```

线性表的定义 数学模型 ADT格式
```
ADT List {

     数据对象: D = {ai | ai属于Elemset, (i = 1, 2,...n, n>=0)}

     数据关系: R = { <ai-1, ai> | ai-1, ai属于D, (i = 2,3,...n)}

     基本操作: lnitList(&L);                // 初始化 构建一个空线性表        
              DestoryList(&L);             // 销毁 线性表消失

  

              ListInsert(&L, i, e);        // 插入 在i前
              ListDelete(&L, i, &e);       
              // 删除 删除前长度为n 删除后长度为n-1 第i个被删除
              ListTraverse(&L, visited()); // Traverse遍历 把每个元素都访问一遍

  

              ClearList(&L);              // 重置线性表为空表 线性表已存在 格式化
              ListEmpty(L);                
              // 判断线性表是否为空 n=0返回TRUE,FALSE
              ListLength(L);               // 求线性表长度

  

              GetElem(L, i, &e);           
              // 用e返回线性表中的第i个元素 1 <= i <= Length
              LocateElem(L, e, compare()); // 查找,定位元素(范围型元素)

  
              PriorElem(L, cur_e, &pre_e); // 求某元素前驱
              NextElem(L, cur_e, &next_e); // 求某元素后继
}ADT List
```

算法三 插入算法 在第i个位置插入一个元素 使长度为n的线性表变成长度为n + 1的
/* 思想:

     1.判断插入位置是否合法

     2.判断顺序表是否已满 已满不能继续插入 返回ERROR

     3.将第n至第i个元素位置依次向后移一位 空出第i个位置

     4.将要插入的新元素e放到第i个位置
*/
```
Status ListInsert_Sq(SqList &L, int i, ElemType e){

     if (i < 1 || i > L.length + 1) return ERROR;   // 判断i是否合法
                   // ⬆下标所表示的数据个数

     if (L.length == MAXSIZE) return ERROR;         // 判断空间是否已满

     for (j = L.length - 1; j >= i-1; j--){
         // ⬆最后一个元素

          L.elem[j + 1] = L.elem[j];                // 将插入位置元素及之后元素后移

     }

     L.elem[i - 1] = e;                             // 将新元素放在第i个位置

	 L.length++;                                    // 增加1格表长

	 return OK;
}
```

/* 计算时间效率  L.elem[j + 1] = L.elem[j];

     若插入在最后 只执行一次 最快 1/(n+1) * 0


     插入在首位置 每个元素都需移动 最慢 1/(n+1) * n


     考虑插入中间某个位置 1/(n+1) * i

     移动次数 n - i + 1

     计算平均移动次数  一共有n+1个考虑位置 (在全部元素的最末端磕插入+每个元素中间可插入)

     每个位置插入的概率是 1/(n+1)

  
     平均移动次数 = 1/(n+1) * n(n+1)/2  =  n/2  (n是线性表中的元素个数)

     则该算法的时间复杂度为 O(n)
*/



算法四 删除元素

/* 线性表长度由n变为n-1

     思想:

     1.判断删除位置i是否合法

     2.将欲删除元素保留在e中 相当于保存在回收站 此步骤可省略

     3.将第i+1至第n位元素依次向前移动一个位置

     4.将表长减1 返回OK
*/
```
Status ListDelete_Sq(SqList &L, int i){

     if(i < 1 || (i > L.length)) return ERROR;  // i值不合法

     for(j = i; j <= L.length - 1; j++){  
// 从要求删除的位置开始 直到最后一个元素 向前移(将要求删除的位置覆盖 即该处元素被删除)      

          L.elem[j - 1] = L.elem[j];            // 将被删除的元素之后的元素前移

     }

     L.length--;                                // 表长减1

     return OK;
}
```

计算时间复杂度

/* L.elem[j - 1] = L.elem[j];

     若删除最后 只执行一次 最快 1/(n-1) * 0

  
     删除第一个 每个元素都需移动 最慢 1/(n-1) * n


     考虑删除中间某个位置 1/(n-1) * i

     移动次数 n - i - 1

  
     计算平均移动次数  一共有n-1个考虑位置 (在全部元素的最末端磕插入+每个元素中间可插入)

     每个位置插入的概率是 1/(n-1)

  
     平均移动次数(期望值) = 1/n * n(n-1)/2  =  n-1/2  (n是线性表中的元素个数)

     则该算法的时间复杂度为 O(n)
*/
