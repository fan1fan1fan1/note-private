##### 前缀和+哈希  

处理连续子数组和问题 求连续子数组和为k、和为0、和能被num整除

```python
def fn(arr, k):
    # 求连续子数组和为k
    from collections import defaultdict
    prefix = 0
    cnt = defaultdict(int)
    cnt[0] = 1   # 初值
    res = 0
    for num in arr:
        prefix += num
        target = prefix - k  # 找和为k(两数之差解法)
        res += cnt.get(target, 0)
        cnt[prefix] += 1
    return res



arr = input()
k = input()  # 和
prefix = 0
dic = {0:1}
res = 0
for x in arr:
    prefix += x
    target = prefix - k
    if target in dic:
         res += dic[target]
    if prefix in dic:
		dic[prefix] += 1
    else:
		dic[prefix] = 1
print(res)
	
```
